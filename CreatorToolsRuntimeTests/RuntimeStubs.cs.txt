    // Unity owns the real implementation. The linked backlog only needs this
    // narrow main-thread dispatch contract, so the harness records every call.
    internal sealed class CreatorToolsInteractionController
    {
        private int availableCapacity;
        private int availableHelpCapacity;

        internal CreatorToolsInteractionController(
            int availableCapacity, int availableHelpCapacity = -1)
        {
            this.availableCapacity = Math.Max(0, availableCapacity);
            this.availableHelpCapacity = availableHelpCapacity < 0
                ? this.availableCapacity
                : Math.Max(0, availableHelpCapacity);
        }

        internal int EnqueueCallCount { get; private set; }
        internal int EnqueuedCount { get; private set; }
        internal int LastDurationSeconds { get; private set; }
        internal int LastCountdownSeconds { get; private set; }
        internal string LastItem { get; private set; }
        internal readonly System.Collections.Generic.List<string>
            EnqueuedItems = new System.Collections.Generic.List<string>();

        internal int StreamQueueAvailableCapacity
        {
            get { return availableCapacity; }
        }

        internal int StreamQueueAvailableCapacityFor(string item)
        {
            return CreatorToolsHelp.Supports(item)
                ? availableHelpCapacity
                : availableCapacity;
        }

        internal int EnqueueStreamInteraction(
            string item,
            string donor,
            string giftImagePath,
            int quantity,
            out string feedbackCode, int durationSeconds = CreatorToolsTimedChallenge.DefaultDuration, int countdownSeconds = CreatorToolsTimedChallenge.DefaultCountdown)
        {
            EnqueueCallCount++;
            LastDurationSeconds = durationSeconds;
            LastCountdownSeconds = countdownSeconds;
            LastItem = item;
            var available = StreamQueueAvailableCapacityFor(item);
            var accepted = Math.Max(
                0, Math.Min(Math.Max(0, quantity), available));
            if (CreatorToolsHelp.Supports(item))
                availableHelpCapacity -= accepted;
            else
                availableCapacity -= accepted;
            EnqueuedCount += accepted;
            for (var i = 0; i < accepted; i++)
                EnqueuedItems.Add(item);
            feedbackCode = accepted > 0 ? "queued" : "queue_full";
            return accepted;
        }
    }
