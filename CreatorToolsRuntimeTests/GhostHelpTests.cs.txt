namespace Gilomx.CupheadBossRoulette
{
    internal static partial class Program
    {
        private static void RunGhostHelpTests()
        {
            Run("Ghost Help keeps FIFO donor credits", delegate
            {
                var queue = new CreatorToolsGhostQueue();
                var first = queue.Enqueue("First helper", "gift-a.png");
                var second = queue.Enqueue("Second helper", "gift-b.png");

                CreatorToolsGhostQueue.Credit credit;
                True(queue.TryDequeue(out credit),
                    "the oldest ghost credit can start");
                Equal(first.Id, credit.Id,
                    "the first redemption starts first");
                Equal("First helper", credit.Donor,
                    "the donor stays attached to the active ghost");
                Equal(1, queue.Count,
                    "the second redemption keeps waiting");
                True(queue.TryDequeue(out credit),
                    "the next ghost can start after the first");
                Equal(second.Id, credit.Id,
                    "the second redemption starts next");
            });

            Run("Ghost Help composes with Half Damage", delegate
            {
                Equal(1f,
                    CreatorToolsGhostDamagePolicy.CalculateMultiplier(
                        false, false),
                    "no modifier keeps normal damage");
                Equal(0.5f,
                    CreatorToolsGhostDamagePolicy.CalculateMultiplier(
                        true, false),
                    "Half Damage still halves damage");
                Equal(2f,
                    CreatorToolsGhostDamagePolicy.CalculateMultiplier(
                        false, true),
                    "Ghost Help doubles total damage");
                Equal(1f,
                    CreatorToolsGhostDamagePolicy.CalculateMultiplier(
                        true, true),
                    "Ghost Help and Half Damage cancel to normal damage");
            });

            Run("Ghost Help keeps a cached frame through player transitions", delegate
            {
                False(CreatorToolsGhostFramePolicy.CanAdoptSourceFrame(
                        true, false, true, 1f),
                    "a disabled transition renderer is not adopted");
                False(CreatorToolsGhostFramePolicy.CanAdoptSourceFrame(
                        true, true, true, 0f),
                    "a fully transparent transition renderer is not adopted");
                True(CreatorToolsGhostFramePolicy.ShouldDisplayCachedFrame(
                        true, false,
                        CreatorToolsGhostMotionPolicy.MaximumOpacity),
                    "the last valid ghost frame stays visible while the player renderer is hidden");
                False(CreatorToolsGhostFramePolicy.MayUseAlternateRenderer(true),
                    "ground jumps cannot replace the full body with one auxiliary Chalice piece");
                True(CreatorToolsGhostFramePolicy.MayUseAlternateRenderer(false),
                    "aircraft transformations may provide a complete alternate renderer");
                Equal(0.44f,
                    CreatorToolsGhostFramePolicy.ResolveGhostAlpha(
                        0f, CreatorToolsGhostMotionPolicy.MaximumOpacity),
                    "player transparency does not erase the ghost");
                False(CreatorToolsGhostFramePolicy.ShouldDisplayCachedFrame(
                        true, true, 0f),
                    "the normal ghost entrance and exit fades can still hide it");
            });

            Run("Ghost Help emerges from and returns to the player", delegate
            {
                Equal(0f,
                    CreatorToolsGhostMotionPolicy.ResolveEntranceOffsetFactor(
                        0f, 0.22f),
                    "the entrance begins inside the player");
                Equal(0.5f,
                    CreatorToolsGhostMotionPolicy.ResolveEntranceOffsetFactor(
                        0.11f, 0.22f),
                    "the entrance uses the mirrored smooth transition");
                Equal(1f,
                    CreatorToolsGhostMotionPolicy.ResolveEntranceOffsetFactor(
                        0.22f, 0.22f),
                    "the entrance reaches its floating offset");
                Equal(1f,
                    CreatorToolsGhostMotionPolicy.ResolveExitOffsetFactor(
                        0.24f, 0.24f),
                    "the exit begins at its floating offset");
                Equal(0f,
                    CreatorToolsGhostMotionPolicy.ResolveExitOffsetFactor(
                        0f, 0.24f),
                    "the exit finishes inside the player");
                Equal(0.44f,
                    CreatorToolsGhostMotionPolicy.ResolveOpacity(
                        1f, 10f, 0.22f, 0.24f),
                    "the visible body is capped at forty-four percent opacity");
            });
        }
    }
}
