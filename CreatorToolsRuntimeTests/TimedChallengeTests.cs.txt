                    CreatorToolsTimedChallenge.NoEx, CreatorToolsTimedChallenge.HalfDamage });

                    var results = new System.Collections.Generic.HashSet<string>();

                        var draws = 0;
                        var selected = CreatorToolsTimedChallenge.ChooseAutomatic(plane, enabled.Contains,
