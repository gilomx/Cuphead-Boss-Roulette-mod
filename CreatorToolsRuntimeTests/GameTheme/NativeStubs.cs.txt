using System;
using System.Linq;
using System.Runtime.CompilerServices;

// Recording native boundaries only. GameTheme's cache, scan timer and retry
// replacement are the unmodified production class linked by this project.
namespace UnityEngine
{
    public class Object
    {
        internal bool Destroyed;
        public string name;
        private static bool IsNull(Object value)
        { return ReferenceEquals(value, null) || value.Destroyed; }
        public static bool operator ==(Object left, Object right)
        {
            if (IsNull(left)) return IsNull(right);
            return !IsNull(right) && ReferenceEquals(left, right);
        }
        public static bool operator !=(Object left, Object right) { return !(left == right); }
        public override bool Equals(object value) { return this == value as Object; }
        public override int GetHashCode() { return RuntimeHelpers.GetHashCode(this); }
        public static void Destroy(Object value) { if (!ReferenceEquals(value, null)) value.Destroyed = true; }
    }
    public sealed class Font : Object { }
    public sealed class Sprite : Object
    {
        public Texture2D texture;
        public Rect textureRect;
    }
    public enum TextureFormat { ARGB32 }
    public enum TextureWrapMode { Repeat, Clamp }
    public enum FilterMode { Point }
    public sealed class Texture2D : Object
    {
        public readonly int width, height;
        public TextureWrapMode wrapMode;
        public FilterMode filterMode;
        public static readonly Texture2D whiteTexture = new Texture2D(1, 1, TextureFormat.ARGB32, false);
        public Texture2D(int width, int height, TextureFormat format, bool mipmaps)
        { this.width = width; this.height = height; }
        public void SetPixels32(Color32[] pixels) { }
        public void SetPixel(int x, int y, Color color) { }
        public void Apply(bool mipmaps, bool release) { }
    }
    public struct Color
    {
        public float r, g, b, a;
        public Color(float r, float g, float b, float a = 1f) { this.r = r; this.g = g; this.b = b; this.a = a; }
        public static Color white { get { return new Color(1f, 1f, 1f, 1f); } }
    }
    public struct Color32
    {
        public byte r, g, b, a;
        public Color32(byte r, byte g, byte b, byte a) { this.r = r; this.g = g; this.b = b; this.a = a; }
        public static implicit operator Color(Color32 value)
        { return new Color(value.r / 255f, value.g / 255f, value.b / 255f, value.a / 255f); }
    }
    public struct Rect
    {
        public float x, y, width, height;
        public float xMax { get { return x + width; } }
        public float yMax { get { return y + height; } }
        public Rect(float x, float y, float width, float height)
        { this.x = x; this.y = y; this.width = width; this.height = height; }
    }
    public static class Mathf
    { public static int Clamp(int value, int minimum, int maximum) { return Math.Max(minimum, Math.Min(maximum, value)); } }
    public static class Time { public static float realtimeSinceStartup; }
    public static class GUI
    {
        public static Color color;
        internal static int SpriteDrawCalls;
        public static void DrawTexture(Rect rect, Texture2D texture) { }
        public static void DrawTextureWithTexCoords(Rect rect, Texture2D texture, Rect uv, bool alpha)
        { SpriteDrawCalls++; }
    }
    public static class Resources
    {
        internal static Object[] Loaded = new Object[0];
        internal static int FontQueries, SpriteQueries;
        internal static void ResetCounters() { FontQueries = SpriteQueries = 0; }
        public static T[] FindObjectsOfTypeAll<T>() where T : Object
        {
            if (typeof(T) == typeof(Font)) FontQueries++;
            if (typeof(T) == typeof(Sprite)) SpriteQueries++;
            return Loaded.OfType<T>().Where(value => value != null).ToArray();
        }
    }
}
