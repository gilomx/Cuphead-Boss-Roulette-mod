using System;

// Only Unity time/lifecycle and the server projection are faked. Battle
// settings, recruitment, scheduler, queue and timed executor are production code.
namespace UnityEngine
{
    public class Object { public static void Destroy(Object value) { } }
    public class GameObject : Object { public void SetActive(bool value) { } }
    public class Component : Object { public GameObject gameObject = new GameObject(); }
    public static class Time { public static float deltaTime; public static float realtimeSinceStartup; }
    public static class Mathf
    {
        public static int CeilToInt(float value) { return (int)Math.Ceiling(value); }
        public static int Max(int a, int b) { return Math.Max(a, b); }
        public static float Max(float a, float b) { return Math.Max(a, b); }
        public static float Clamp01(float value) { return Math.Max(0f, Math.Min(1f, value)); }
        public static bool Approximately(float a, float b) { return Math.Abs(a - b) < 0.00001f; }
    }
    public static class Random
    {
        public static int Range(int minimum, int maximum) { return minimum; }
        public static float Range(float minimum, float maximum) { return minimum; }
    }
}

internal enum Levels { Ground, Plane, DicePalaceMain, DicePalaceBooze }
internal sealed class Level
{
    internal enum Type { Battle }
    internal static Level Current;
    internal Type LevelType;
    internal bool Ending;
    internal Levels CurrentLevel;
    internal int Id;
    internal int GetInstanceID() { return Id; }
}

namespace Gilomx.CupheadBossRoulette
{
    internal sealed class CreatorToolsServer
    {
        internal bool IsRunning { get { return true; } }
        internal string State;
        internal int BattlePublications;
        internal string TapState;
        internal int TapPublications;
        internal void SetPeskyBattleState(string state) { State = state; BattlePublications++; }
        internal void SetTapFarmingState(string state) { TapState = state; TapPublications++; }
        internal void ApplyPeskyBattleMainThreadActions(Action apply) { apply(); }
        internal bool TryTakePeskyBattleCommand(out string query) { query = null; return false; }
    }
}
