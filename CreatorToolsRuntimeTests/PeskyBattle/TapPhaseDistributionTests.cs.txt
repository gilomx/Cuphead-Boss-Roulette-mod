    private static void TestGenericDistribution()
    {
        var directory = Path.Combine(Path.GetTempPath(), "pichi-phase-" + Guid.NewGuid().ToString("N"));
        Directory.CreateDirectory(directory);
        var taps = new CreatorToolsTapFarmingController(Path.Combine(directory, "plugin.cfg"),
            new CreatorToolsLiveEventsCoordinator(), null, null);
        var server = new CreatorToolsServer();
        try
        {
            taps.ProcessBackgroundCommand("action=activate&tapsPerConversion=1&healthPointsPerConversion=1");
            var properties = new Properties();
            var level = new Level { CurrentLevel = Levels.Ground, Id = 400 };
            taps.OnLevelStarted(level);
            Deposit(taps, 1000);
            Hit(taps, properties, 201, 1);
            Near(800, Reserve(taps, server));
            // Phase 2 begins before Time.realtimeSinceStartup changes.
            properties.stateIndex = 1; properties.CurrentHealth = 800;
            Hit(taps, properties, 100, 0);
            Near(700, Reserve(taps, server));
            Deposit(taps, 300); // 150 for phase 2, 150 for phase 3.
            Hit(taps, properties, 451, 1);
            Near(550, Reserve(taps, server));
            properties.stateIndex = 2; properties.CurrentHealth = 400;
            Hit(taps, properties, 100, 0);
            Near(450, Reserve(taps, server));
            taps.OnLevelDefeated(level);
            Near(450, Reserve(taps, server));
            using (var retry = JsonDocument.Parse(server.TapState))
                Near(450, retry.RootElement.GetProperty("counters").GetProperty("bankedTaps").GetDouble());
            properties.stateIndex = 0; properties.CurrentHealth = 1000;
            taps.OnLevelStarted(new Level { CurrentLevel = Levels.Ground, Id = 401 });
            Hit(taps, properties, 91, 1); // Remaining 450 repartitions 90/180/180.
            Near(360, Reserve(taps, server));
            Console.WriteLine("PASS controller phase distribution, new taps, immediate phase changes and retry conservation");
        }
        finally
        {
            taps.ProcessBackgroundCommand("action=off");
            Directory.Delete(directory, true);
        }
    }

    private static void TestDicePalaceSharedReserveAndBoardLifecycle()
    {
        using (var fixture = new Fixture())
        {
            var board = fixture.Start(Levels.DicePalaceMain, 500);
            Deposit(fixture.Taps, 400);
            AssertLifecycle(fixture, "transition", 1);
            AssertDiceReserve(fixture, 400);
            AssertEffective(fixture, false, 400);
            fixture.MainState(board, false);
            fixture.Taps.ObserveBossDamage(Arena(900));
            AssertLifecycle(fixture, "transition", 1);
            AssertDiceReserve(fixture, 400);
            AssertEffective(fixture, false, 400);

            // Leaving the board has no miniboss PreWin notification.
            fixture.LeaveBoard(board);
            var first = fixture.Start(Levels.DicePalaceChips, 501);
            var properties = Arena(1000);
            Hit(fixture.Taps, properties, 150, 0);
            AssertDiceReserve(fixture, 250); // More than the old 100-per-arena budget.
            AssertAttemptAndArena(fixture, 1, 1000);
            AssertEffective(fixture, true, 1250, 1000, 1.25);
            fixture.Handoff(first, endedFirst: true);
            AssertEffective(fixture, true, 1250, 1000, 1.25);
            board = fixture.Start(Levels.DicePalaceMain, 502);
            AssertLifecycle(fixture, "transition", 1);
            fixture.MainState(board, false);
            fixture.Taps.ObserveBossDamage(Arena(900));
            AssertDiceReserve(fixture, 250);
            AssertEffective(fixture, true, 1250, 1000, 1.25);
            Deposit(fixture.Taps, 40);
            AssertDiceReserve(fixture, 290);
            AssertEffective(fixture, true, 1290, 1000, 1.29);

            fixture.LeaveBoard(board);
            var second = fixture.Start(Levels.DicePalaceChips, 503);
            properties = Arena(500);
            Hit(fixture.Taps, properties, 291, 1);
            fixture.Refresh(properties);
            AssertDiceReserve(fixture, 0);
            AssertAttemptAndArena(fixture, 1, 500);
            AssertEffective(fixture, true, 499, 500, .998);

            fixture.Handoff(second);
            board = fixture.Start(Levels.DicePalaceMain, 504);
            AssertLifecycle(fixture, "transition", 1);
            AssertEffective(fixture, true, 499, 500, .998);
            fixture.LeaveBoard(board);
            var third = fixture.Start(Levels.DicePalaceChips, 505);
            properties = Arena(700);
            Deposit(fixture.Taps, 60);
            Hit(fixture.Taps, properties, 60, 0);
            AssertDiceReserve(fixture, 0);

            fixture.Handoff(third);
            board = fixture.Start(Levels.DicePalaceMain, 506);
            fixture.MainState(board, false);
            AssertLifecycle(fixture, "transition", 1);
            Deposit(fixture.Taps, 40);
            AssertDiceReserve(fixture, 40);
            AssertEffective(fixture, true, 740, 700, 1.0571);
            fixture.Taps.ObserveBossDamage(Arena(900));
            AssertDiceReserve(fixture, 40);
            AssertEffective(fixture, true, 740, 700, 1.0571);

            // An extra miniboss spends new taps from that same shared balance.
            fixture.LeaveBoard(board);
            var extra = fixture.Start(Levels.DicePalaceChips, 507);
            properties = Arena(400);
            Hit(fixture.Taps, properties, 41, 1);
            fixture.Refresh(properties);
            AssertDiceReserve(fixture, 0);
            AssertEffective(fixture, true, 399, 400, .9975);

            fixture.Handoff(extra);
            var main = fixture.Start(Levels.DicePalaceMain, 508);
            AssertLifecycle(fixture, "transition", 1);
            fixture.MainState(main, false);
            fixture.Taps.ObserveBossDamage(Arena(900));
            AssertDiceReserve(fixture, 0);
            AssertEffective(fixture, true, 399, 400, .9975);
            fixture.MainState(board, true); // A stale board instance cannot activate Main.
            AssertLifecycle(fixture, "transition", 1);

            // Main becomes the final arena in this same Level instance only
            // when its native state reports Battling.
            fixture.MainState(main, true);
            AssertLifecycle(fixture, "active", 1);
            properties = Arena(900);
            Hit(fixture.Taps, properties, 100, 100);
            AssertDiceReserve(fixture, 0); // Nothing was protected for the final.
            AssertAttemptAndArena(fixture, 1, 900);
            Deposit(fixture.Taps, 25);
            Hit(fixture.Taps, properties, 26, 1);
            fixture.Refresh(properties);
            AssertDiceReserve(fixture, 0);
            Near(799, properties.CurrentHealth);
            fixture.MainState(main, true);
            AssertLifecycle(fixture, "active", 1);
            AssertEffective(fixture, true, 799, 900, .8878);

            fixture.Taps.OnLevelPreWin(main);
            AssertLifecycle(fixture, "completed", 0);
            using (var state = fixture.State())
            {
                Near(565, state.RootElement.GetProperty("lastRound").GetProperty("spentHealth").GetDouble());
                Near(1, state.RootElement.GetProperty("lastRound").GetProperty("attempts").GetInt32());
            }
            Console.WriteLine("PASS Dice Palace shared reserve, board health retention, native Ended-before-PreWin, extra arenas and final Battling marker");
        }
    }

    {

            fixture.Handoff(first);
            board = fixture.Start(Levels.DicePalaceMain, 802);
            fixture.Taps.OnMapEntered();
            AssertLifecycle(fixture, "collecting", 1);
            Near(275, Reserve(fixture.Taps, fixture.Server));

            var retryBoard = fixture.Start(Levels.DicePalaceMain, 803);
            AssertLifecycle(fixture, "transition", 2);
            AssertEffective(fixture, false, 275);
            fixture.MainState(board, true);
            AssertLifecycle(fixture, "transition", 2);
            fixture.LeaveBoard(retryBoard);

            Hit(fixture.Taps, Arena(600), 300, 25);
            AssertDiceReserve(fixture, 0);
            AssertAttemptAndArena(fixture, 2, 600);

            Deposit(fixture.Taps, 80);
            fixture.Taps.OnLevelDefeated(retry);
            fixture.Taps.OnLevelEnded(retry);
            fixture.Taps.OnLevelPreWin(retry);
            AssertLifecycle(fixture, "collecting", 2);
            Near(80, Reserve(fixture.Taps, fixture.Server));
            retryBoard = fixture.Start(Levels.DicePalaceMain, 805);
            AssertLifecycle(fixture, "transition", 3);
            AssertEffective(fixture, false, 80);
            fixture.LeaveBoard(retryBoard);
            fixture.Start(Levels.DicePalaceChips, 806);
            Hit(fixture.Taps, Arena(600), 81, 1);
            AssertDiceReserve(fixture, 0);
            AssertAttemptAndArena(fixture, 3, 600);
            Console.WriteLine("PASS map exit and actual Dice Palace defeat close the chain; new attempts keep only unspent shared reserve");
        }
    }

    private static void TestDicePalaceGlobalProgressAndIndependentTapReserve()
    {
        using (var fixture = new Fixture())
        {
            var board = fixture.Start(Levels.DicePalaceMain, 1000);
            AssertBattleProgress(fixture, "encounter", true, 0, true);
            fixture.MainState(board, false);
            fixture.LeaveBoard(board);
            var first = fixture.Start(Levels.DicePalaceChips, 1001);
            var properties = Arena(1000);
            Hit(fixture.Taps, properties, 400, 400);
            fixture.Refresh(properties);
            AssertBattleProgress(fixture, "encounter", true, .1, true);
            Near(600, properties.CurrentHealth);
            fixture.Handoff(first, endedFirst: true);
            AssertBattleProgress(fixture, "encounter", true, .25, true);

            board = fixture.Start(Levels.DicePalaceMain, 1002);
            fixture.MainState(board, false);
            fixture.Taps.ObserveBossDamage(Arena(900));
            AssertBattleProgress(fixture, "encounter", true, .25, true);
            Deposit(fixture.Taps, 50);
            AssertDiceReserve(fixture, 50);
            AssertBattleProgress(fixture, "encounter", true, .25, true);

            fixture.LeaveBoard(board);
            var second = fixture.Start(Levels.DicePalaceChips, 1003);
            properties = Arena(500);
            // The first 50 damage is real reserve absorption. Only the native
            // 250 remainder advances the second segment halfway to 50%.
            Hit(fixture.Taps, properties, 300, 250);
            fixture.Refresh(properties);
            AssertDiceReserve(fixture, 0);
            AssertBattleProgress(fixture, "encounter", true, .375, true);
            Deposit(fixture.Taps, 20);
            AssertBattleProgress(fixture, "encounter", true, .375, true);
            Hit(fixture.Taps, properties, 20, 0);
            fixture.Refresh(properties);
            AssertDiceReserve(fixture, 0);
            Near(250, properties.CurrentHealth);
            AssertBattleProgress(fixture, "encounter", true, .375, true);
            fixture.Handoff(second);
            AssertBattleProgress(fixture, "encounter", true, .5, true);

            board = fixture.Start(Levels.DicePalaceMain, 1004);
            fixture.MainState(board, false);
            AssertBattleProgress(fixture, "encounter", true, .5, true);
            fixture.LeaveBoard(board);
            var third = fixture.Start(Levels.DicePalaceChips, 1005);
            properties = Arena(700);
            Hit(fixture.Taps, properties, 350, 350);
            fixture.Refresh(properties);
            AssertBattleProgress(fixture, "encounter", true, .625, true);
            fixture.Handoff(third);
            AssertBattleProgress(fixture, "encounter", true, .75, true);

            board = fixture.Start(Levels.DicePalaceMain, 1006);
            AssertBattleProgress(fixture, "encounter", true, .75, true);
            fixture.LeaveBoard(board);
            var extra = fixture.Start(Levels.DicePalaceChips, 1007);
            properties = Arena(400);
            Hit(fixture.Taps, properties, 399, 399);
            fixture.Refresh(properties);
            AssertBattleProgress(fixture, "encounter", true, .75, true);
            fixture.Handoff(extra);
            AssertBattleProgress(fixture, "encounter", true, .75, true);

            var main = fixture.Start(Levels.DicePalaceMain, 1008);
            fixture.MainState(main, false);
            fixture.Taps.ObserveBossDamage(Arena(900));
            AssertLifecycle(fixture, "transition", 1);
            AssertBattleProgress(fixture, "encounter", true, .75, true);
            // The board and final fight use this same native Level instance.
            fixture.MainState(main, true);
            AssertLifecycle(fixture, "active", 1);
            AssertBattleProgress(fixture, "encounter", true, .75, true);
            properties = Arena(900);
            Hit(fixture.Taps, properties, 450, 450);
            fixture.Refresh(properties);

            AssertBattleProgress(fixture, "encounter", true, 0, true);
            fixture.LeaveBoard(board);

            Hit(fixture.Taps, properties, 300, 300);
            fixture.Refresh(properties);
            AssertBattleProgress(fixture, "encounter", true, .125, true);

            AssertBattleProgress(fixture, "boss", false, 0, false);

            AssertBattleProgress(fixture, "encounter", true, 0, true);

    private static void TestFailedReaderPreservesNativeDamage()
    {
        using (var fixture = new Fixture())
        {
            fixture.Start(Levels.Ground, 600);
            Deposit(fixture.Taps, 1000);
            var properties = new FailingProperties();
            var damage = 100f;
            Check(!fixture.Taps.PrepareBossDamage(properties, ref damage), "valid initial metadata must absorb");
            Near(0, damage);
            AssertBalances(fixture, 900, 100, 400, 400);

            fixture.Taps.OnPhaseTransition();
            // This models a temporarily unreadable native state while the
            // last good phase map and reserve ledger still exist.
            properties.stateIndex = "unreadable-native-state";
            UnityEngine.Time.realtimeSinceStartup += 0.2f;
            damage = 12f;
            Check(fixture.Taps.PrepareBossDamage(properties, ref damage), "failed phase read must preserve the native callback");
            Near(12, damage);
            properties.CurrentHealth -= damage;
            fixture.Taps.ObserveBossDamage(properties);
            AssertBalances(fixture, 900, 100, 400, 400);
            using (var state = fixture.State())
                Check(state.RootElement.GetProperty("phase").GetString() == "active",
                    "a failed reader must still resume the active lifecycle after the transition");

            properties.stateIndex = 1;
            properties.CurrentHealth = 800;
            // No clock advancement: recovery must use the current phase even
            // while the expensive full snapshot remains throttled.
            damage = 451f;
            Check(fixture.Taps.PrepareBossDamage(properties, ref damage), "recovered phase must admit only the unabsorbed remainder");
            Near(1, damage);
            properties.CurrentHealth -= damage;
            AssertBalances(fixture, 450, 0, 0, 450);
            Near(799, properties.CurrentHealth);
            Console.WriteLine("PASS failed native phase reads leave damage/reserve intact, resume lifecycle and recover within the snapshot throttle");
        }
    }

    private static void TestVictoryStartsNewBalance()
    {
        using (var fixture = new Fixture())
        {
            var first = fixture.Start(Levels.Ground, 700);
            Deposit(fixture.Taps, 1000);
            Hit(fixture.Taps, new Properties(), 150, 0);
            fixture.Taps.OnLevelDefeated(first);
            AssertBalances(fixture, 850); // No phase projection while collecting.
            var retry = fixture.Start(Levels.Ground, 701);
            Hit(fixture.Taps, new Properties(), 160, 0);
            AssertBalances(fixture, 690, 10, 340, 340);
            fixture.Taps.OnLevelPreWin(first); // Late notification from a dead attempt.
            Near(690, Reserve(fixture.Taps, fixture.Server));
            fixture.Taps.OnLevelPreWin(retry);
            using (var won = fixture.State())
            {
                var root = won.RootElement;
                foreach (var field in new[] { "totalTaps", "bankedTaps", "unconvertedTaps", "convertedHealth", "reserveHealth", "spentHealth" })
                    Near(0, root.GetProperty("counters").GetProperty(field).GetDouble());
                Near(1000, root.GetProperty("lastRound").GetProperty("totalTaps").GetInt64());
                Near(310, root.GetProperty("lastRound").GetProperty("spentHealth").GetDouble());
                Near(2, root.GetProperty("lastRound").GetProperty("attempts").GetInt32());
            }
            Deposit(fixture.Taps, 9);
            fixture.Taps.OnMapEntered();
            fixture.Taps.Update(fixture.Server, true); // The ending Level can still be exposed.
            using (var collecting = fixture.State())
            {
                Check(collecting.RootElement.GetProperty("phase").GetString() == "collecting", "completed native level must not reattach");
                Near(9, collecting.RootElement.GetProperty("counters").GetProperty("totalTaps").GetInt64());
                Near(9, collecting.RootElement.GetProperty("counters").GetProperty("reserveHealth").GetDouble());
            }
            fixture.Start(Levels.Ground, 702);
            Hit(fixture.Taps, new Properties(), 1, 0);
            Near(8, Reserve(fixture.Taps, fixture.Server));
            Console.WriteLine("PASS retry never refunds consumed reserve; victory clears counters and starts a fresh bank without losing results-screen likes");
        }
    }

    private static Properties Arena(float total)
    {
        return new Properties { CurrentHealth = total, TotalHealth = total,
            states = new[] { new State("Main", 1) } };
    }

    private static void AssertBalances(Fixture fixture, double reserve, params double[] balances)
    {
        using (var state = fixture.State())
        {
            Near(reserve, state.RootElement.GetProperty("counters").GetProperty("reserveHealth").GetDouble());
            var phases = state.RootElement.GetProperty("phases");
            Check(phases.GetArrayLength() == balances.Length, "logical phase count changed");
            for (var index = 0; index < balances.Length; index++)
                Near(balances[index], phases[index].GetProperty("reserveHealth").GetDouble());
        }
    }

    private static void AssertAttemptAndArena(Fixture fixture, int attempt, double nativeTotal)
    {
        using (var state = fixture.State())
        {
            Near(attempt, state.RootElement.GetProperty("attempt").GetInt32());
            var effective = state.RootElement.GetProperty("effectiveHealth");
            Check(effective.GetProperty("scope").GetString() == "arena", "Dice health percent must identify its per-arena denominator");
            Near(nativeTotal, effective.GetProperty("total").GetDouble());
        }
    }

    private static void AssertLifecycle(Fixture fixture, string phase, int attempt)
    {
        using (var state = fixture.State())
        {
            Check(state.RootElement.GetProperty("phase").GetString() == phase,
                "Expected lifecycle " + phase + ", got " + state.RootElement.GetProperty("phase").GetString());
            Near(attempt, state.RootElement.GetProperty("attempt").GetInt32());
        }
    }

    private static void AssertDiceReserve(Fixture fixture, double reserve)
    {
        // Dice's four route phases describe progress, never independent HP budgets.
        AssertBalances(fixture, reserve, 0, 0, 0, 0);
    }

    private static void AssertEffective(Fixture fixture, bool available,
        double current = 0, double total = 0, double ratio = 0)
    {
        using (var state = fixture.State())
        {
            var effective = state.RootElement.GetProperty("effectiveHealth");
            Check(effective.GetProperty("available").GetBoolean() == available,
                "effective HP availability changed during a board handoff");
            Near(current, effective.GetProperty("current").GetDouble());
            Near(total, effective.GetProperty("total").GetDouble());
            Near(ratio, effective.GetProperty("ratio").GetDouble());
        }
    }

    private static void AssertBattleProgress(Fixture fixture, string scope,
        bool available, double ratio, bool estimated)
    {
        using (var state = fixture.State())
        {
            var progress = state.RootElement.GetProperty("battleProgress");
            Check(progress.GetProperty("scope").GetString() == scope,
                "battle progress scope must distinguish a whole Dice Palace encounter from one boss");
            Check(progress.GetProperty("available").GetBoolean() == available,
                "battle progress must remain visible across Dice Palace handoffs and clear while collecting");
            Near(ratio, progress.GetProperty("ratio").GetDouble());
            Check(progress.GetProperty("estimated").GetBoolean() == estimated,
                "Dice Palace's four route segments must identify their global progress as an estimate");
            Near(ratio, state.RootElement.GetProperty("overallProgress").GetDouble());
        }
    }

    private sealed class Fixture : IDisposable
    {
        private readonly string directory = Path.Combine(Path.GetTempPath(), "pichi-phase-integration-" + Guid.NewGuid().ToString("N"));
        private readonly Level previousLevel = Level.Current;
        private readonly float previousTime = UnityEngine.Time.realtimeSinceStartup;
        internal readonly CreatorToolsTapFarmingController Taps;
        internal readonly CreatorToolsServer Server = new CreatorToolsServer();

        internal Fixture()
        {
            Directory.CreateDirectory(directory);
            UnityEngine.Time.realtimeSinceStartup = 0f;
            Taps = new CreatorToolsTapFarmingController(Path.Combine(directory, "plugin.cfg"),
                new CreatorToolsLiveEventsCoordinator(), null, null);
            Taps.ProcessBackgroundCommand("action=activate&tapsPerConversion=1&healthPointsPerConversion=1");
        }
        internal Level Start(Levels arena, int id)
        {
            var level = new Level { CurrentLevel = arena, Id = id };
            Level.Current = level;
            Taps.OnLevelStarted(level);
            return level;
        }
        internal void Handoff(Level previous, bool endedFirst = false)
        {
            int currentAttempt;
            using (var state = State())
                currentAttempt = state.RootElement.GetProperty("attempt").GetInt32();
            if (endedFirst)
            {
                Taps.OnLevelEnded(previous);
                AssertLifecycle(this, "transition", currentAttempt);
            }
            Taps.OnLevelPreWin(previous);
            AssertLifecycle(this, "transition", currentAttempt);
            int completedPhaseIndex;
            using (var state = State())
                completedPhaseIndex = state.RootElement.GetProperty("phaseIndex").GetInt32();
            // The overlay bridge calls PreWin then Ended; native PreWin can
            // subsequently report the same win again.
            Taps.OnLevelEnded(previous);
            Taps.OnLevelPreWin(previous);
            AssertLifecycle(this, "transition", currentAttempt);
            using (var state = State())
                Near(completedPhaseIndex, state.RootElement.GetProperty("phaseIndex").GetInt32());
        }
        internal void LeaveBoard(Level previous)
        {
            MainState(previous, false);
            Taps.OnLevelEnded(previous);
        }
        internal void MainState(Level main, bool battling)
        {
            Taps.ObserveDicePalaceMainState(main.GetInstanceID(), battling);
        }
        internal void Refresh(object properties)
        {
            UnityEngine.Time.realtimeSinceStartup += 0.2f;
            Taps.ObserveBossDamage(properties);
        }
        internal JsonDocument State()
        {
            Taps.PublishState(Server);
            return JsonDocument.Parse(Server.TapState);
        }
        public void Dispose()
        {
            Taps.ProcessBackgroundCommand("action=off");
            Level.Current = previousLevel;
            UnityEngine.Time.realtimeSinceStartup = previousTime;
            Directory.Delete(directory, true);
        }
    }

        public float CurrentHealth { get; set; } = 1000;

    private static void Deposit(CreatorToolsTapFarmingController taps, int count)
    {
        taps.ObserveStreamEvent(new CreatorToolsStreamEvent { Platform = "tiktok", Type = "like", Count = count });
    }
    private static void Hit(CreatorToolsTapFarmingController taps, Properties properties, float hit, float remaining)
    {
        var damage = hit;
        var passes = taps.PrepareBossDamage(properties, ref damage);
        Near(remaining, damage);
        if (passes != (remaining > 0)) throw new Exception("native damage admission mismatch");
        if (passes) properties.CurrentHealth -= damage;
        taps.ObserveBossDamage(properties);
    }
    private static double Reserve(CreatorToolsTapFarmingController taps, CreatorToolsServer server)
    {
        taps.PublishState(server);
        using (var state = JsonDocument.Parse(server.TapState))
            return state.RootElement.GetProperty("counters").GetProperty("reserveHealth").GetDouble();
    }
    private static void Near(double expected, double actual)
    {
        if (double.IsNaN(actual) || double.IsInfinity(actual) ||
            Math.Abs(expected - actual) > 0.0001)
            throw new Exception("Expected " + expected + ", got " + actual);
    }
    private static void Check(bool condition, string message)
    {
        if (!condition) throw new Exception(message);
    }

    {
        public float CurrentHealth { get; set; } = 1000;
        public float TotalHealth = 1000;
        public int stateIndex;
        public State[] states = { new State("A", 1), new State("B", .8f), new State("C", .4f) };
    }

    {
        public string stateName;
        public float healthTrigger;
        public State(string name, float trigger) { stateName = name; healthTrigger = trigger; }
    }
