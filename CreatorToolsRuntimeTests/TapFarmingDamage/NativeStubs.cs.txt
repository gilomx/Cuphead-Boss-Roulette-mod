// The real bridge, controller, reserve arithmetic and progress reader are
// linked into this harness. Only Unity, Harmony patch registration and native
// boss property types are substituted; tests invoke the registered patches

namespace HarmonyLib
{
    public sealed class HarmonyMethod
    {
        internal readonly MethodInfo Method;
        public HarmonyMethod(MethodInfo method) { Method = method; }
    }

    public sealed class Harmony
    {
        internal readonly List<PatchRecord> Patches = new List<PatchRecord>();
        internal Func<MethodBase, bool> RejectPatch;
        public void Patch(MethodBase original, HarmonyMethod prefix = null,
            HarmonyMethod postfix = null, HarmonyMethod finalizer = null,
            HarmonyMethod transpiler = null)
        {
            if (RejectPatch != null && RejectPatch(original))
                throw new InvalidOperationException("simulated unsupported native helper");
            Patches.Add(new PatchRecord { Original = original,
                Prefix = prefix == null ? null : prefix.Method,
                Postfix = postfix == null ? null : postfix.Method,
                Finalizer = finalizer == null ? null : finalizer.Method,
                Transpiler = transpiler == null ? null : transpiler.Method });
        }
    }

    internal sealed class PatchRecord
    {
        internal MethodBase Original;
        internal MethodInfo Prefix, Postfix, Finalizer, Transpiler;
    }

    public static class AccessTools
    {
        public static FieldInfo Field(Type type, string name)
        {
            for (; type != null; type = type.BaseType)
            {
                var result = type.GetField(name, BindingFlags.Instance | BindingFlags.Public |
                    BindingFlags.NonPublic | BindingFlags.DeclaredOnly);
                if (result != null) return result;
            }
            return null;
        }
        public static MethodInfo Method(Type type, string name, Type[] parameters = null)
        {
            const BindingFlags flags = BindingFlags.Instance | BindingFlags.Static |
                BindingFlags.Public | BindingFlags.NonPublic;
            return parameters == null ? type.GetMethod(name, flags) :
                type.GetMethod(name, flags, null, parameters, null);
        }
    }
    public sealed class CodeInstruction
    {
        public OpCode opcode;
        public object operand;
        public readonly List<Label> labels = new List<Label>();
        public readonly List<object> blocks = new List<object>();
        public CodeInstruction(OpCode opcode, object operand = null)
        { this.opcode = opcode; this.operand = operand; }
    }
}

internal class AbstractLevelProperties<TState, TName, TDifficulty>
{
    public float CurrentHealth { get; private set; } = 300f;
    public float TotalHealth = 300f;
    public int stateIndex;
    public RobotState[] states = new[]
    {
        new RobotState { stateName = RobotStateName.Components, healthTrigger = 1f },
        new RobotState { stateName = RobotStateName.Flying, healthTrigger = 0.5f }
    };
    // This callback routes the synthetic 1-damage loop through the real
    // bridge's patches without requiring runtime method rewriting in tests.
    internal Action<float> RouteDamage;
    public void DealDamage(float damage)
    {
        CurrentHealth = Math.Max(0f, CurrentHealth - damage);
        while (stateIndex + 1 < states.Length &&
            CurrentHealth <= TotalHealth * states[stateIndex + 1].healthTrigger + 0.0001f)
            stateIndex++;
    }
    public void DealDamageToNextNamedState()
    {
        var initialName = states[stateIndex].stateName;
        for (var i = 0; i < TotalHealth && states[stateIndex].stateName == initialName; i++)
            RouteDamage(1f);
    }
}

internal sealed class RobotLevelRobot

    private RobotLevelRobotChest chest;
    private RobotLevelRobotHatch hatch;
    internal RobotLevelRobot(RobotLevelRobotHead head, RobotLevelRobotChest chest, RobotLevelRobotHatch hatch)
    { this.head = head; this.chest = chest; this.hatch = hatch; }
}
internal class RobotLevelRobotBodyPart
{
    protected LevelProperties.Robot properties;
    protected float[] currentHealth = new float[2];
    internal void Init(LevelProperties.Robot value, float primary, float secondary)
    { properties = value; currentHealth[0] = primary; currentHealth[1] = secondary; }
    internal float Health(int index) { return currentHealth[index]; }
    internal void SetHealth(int index, float value) { currentHealth[index] = value; }
    protected virtual void OnDamageTaken(DamageDealer.DamageInfo info)
    {
        float old = currentHealth[0];
        currentHealth[0] -= info.damage;
        GC.KeepAlive(old);
    }
}
internal sealed class RobotLevelRobotHead : RobotLevelRobotBodyPart { }
internal sealed class RobotLevelRobotHatch : RobotLevelRobotBodyPart
{
    protected override void OnDamageTaken(DamageDealer.DamageInfo info) { base.OnDamageTaken(info); }
}
internal sealed class RobotLevelRobotChest : RobotLevelRobotBodyPart
{
    protected override void OnDamageTaken(DamageDealer.DamageInfo info)
    {
        float old = currentHealth[1];
        currentHealth[1] -= info.damage;
        GC.KeepAlive(old);
    }
}
