        private static void TestRetiredFloatingControls()
        {
            using (var scope = new TestScope("retired-floating-controls"))
            {
                var controller = scope.CreateOverlayComposer(null);
                Equal(200, controller.ProcessConfigCommand(JsonSerializer.Serialize(new {
                    operation = "update", expectedRevision = 0, profileId = "vertical", componentId = "interactions",
                    x = 50, y = 998, width = 960, height = 280, interactionsSize = 79
                })).StatusCode, "save the confirmed scene geometry");
                var expected = JsonNode.Parse(controller.GetConfigState())["profiles"];
                var path = Path.Combine(scope.ConfigDirectory, "mx.gilomx.cuphead.bossroulette.overlay-composer.json");
                var legacy = JsonNode.Parse(File.ReadAllText(path));
                var component = legacy["profiles"][0]["components"].AsArray().First(item => item["id"].GetValue<string>() == "interactions");
                component["interactionsFloatingOverlap"] = 29;
                component["interactionsFloatingGap"] = 35;
                var legacyText = legacy.ToJsonString();
                File.WriteAllText(path, legacyText);
                using (var state = JsonDocument.Parse(scope.CreateOverlayComposer(null).GetConfigState()))
                {
                    True(JsonNode.DeepEquals(expected, JsonNode.Parse(state.RootElement.GetProperty("profiles").GetRawText())),
                        "retiring controls preserves all explicit scene values");
                    var current = FindComposerComponent(state.RootElement, "vertical", "interactions");
                    False(current.TryGetProperty("interactionsFloatingOverlap", out _), "overlap is no longer an editable scene field");
                    False(current.TryGetProperty("interactionsFloatingGap", out _), "gap is no longer an editable scene field");
                }
                Equal(legacyText, File.ReadAllText(path), "loading the confirmed scene preserves its original file");
            }
        }

        private static void TestInteractionsOverlayBattleVisibility()
        {
            using (var scope = new TestScope("interactions-battle-visibility"))
            {
                var controller = scope.CreateOverlayComposer(null);
                using (var state = JsonDocument.Parse(controller.GetConfigState()))
                    True(state.RootElement.GetProperty("interactionsBattleOnly").GetBoolean(), "battle visibility is enabled by default");
                Equal(200, controller.ProcessConfigCommand(JsonSerializer.Serialize(new {
                    operation = "interactions_visibility", expectedRevision = 0, profileId = "vertical", interactionsBattleOnly = false
                })).StatusCode, "global visibility changes in one atomic revision");
                controller = scope.CreateOverlayComposer(null);
                var before = controller.GetConfigState();
                using (var state = JsonDocument.Parse(before))
                    False(state.RootElement.GetProperty("interactionsBattleOnly").GetBoolean(), "visibility survives a restart");
                foreach (var invalid in new object[] { "true", 1, null })
                {
                    Equal(400, controller.ProcessConfigCommand(JsonSerializer.Serialize(new {
                        operation = "interactions_visibility", expectedRevision = 1, profileId = "vertical", interactionsBattleOnly = invalid
                    })).StatusCode, "invalid visibility is rejected");
                    Equal(before, controller.GetConfigState(), "invalid visibility preserves both scenes and revision");
                }
                Equal(409, controller.ProcessConfigCommand(JsonSerializer.Serialize(new {
                    operation = "interactions_visibility", expectedRevision = 0, profileId = "horizontal", interactionsBattleOnly = true
                })).StatusCode, "stale visibility update is rejected");
                Equal(200, controller.ProcessConfigCommand(JsonSerializer.Serialize(new {
                    operation = "reset", expectedRevision = 1, profileId = "horizontal"
                })).StatusCode, "profile reset succeeds");
                using (var state = JsonDocument.Parse(controller.GetConfigState()))
                    False(state.RootElement.GetProperty("interactionsBattleOnly").GetBoolean(), "resetting a profile preserves global visibility");
                var path = Path.Combine(scope.ConfigDirectory, "mx.gilomx.cuphead.bossroulette.overlay-composer.json");
                var legacy = JsonNode.Parse(File.ReadAllText(path));
                legacy.AsObject().Remove("interactionsBattleOnly");
                File.WriteAllText(path, legacy.ToJsonString());
                using (var state = JsonDocument.Parse(scope.CreateOverlayComposer(null).GetConfigState()))
                {
                    True(state.RootElement.GetProperty("interactionsBattleOnly").GetBoolean(), "legacy scenes gain battle visibility");
                    Equal(legacy["profiles"].ToJsonString(), JsonNode.Parse(state.RootElement.GetProperty("profiles").GetRawText()).ToJsonString(),
                        "visibility migration preserves every saved scene value");
                }
            }

        private static void TestInteractionsOverlayDefaults()
        {
            var settings = CreatorToolsOverlayComposerSettings.CreateDefaults("unused-overlay-settings.json", null);
            foreach (var profile in settings.Profiles)
            {
                var vertical = profile.Id == "vertical";
                var component = profile.FindComponent("interactions");
                Equal(vertical, component.Enabled, "default activation matches each saved profile");
                Equal("floating", component.InteractionsTemplate, "floating template default");
                Equal("strip", component.InteractionsLayout, "saved profiles use strips");
                Equal(!vertical, component.InteractionsTikTok, "TikTok follows saved profile");
                True(component.InteractionsTwitch, "Twitch follows saved profile");
                True(component.InteractionsYouTube, "YouTube follows saved profile");
                True(component.InteractionsAdaptive, "adaptive sizes default");
                Equal(8, component.InteractionsPageSeconds, "page timing default");
                Equal(vertical ? 80 : 40, component.InteractionsSpeed, "saved strip speed default");
                Equal(vertical ? "#662effa6" : "#23262bee", component.InteractionsCardColor, "saved card color includes opacity");
                Equal(vertical ? 80 : 18, component.InteractionsCardNoise, "saved texture strength default");
                Equal(vertical ? 96 : 0, component.InteractionsSize, "saved size follows profile");
                Equal(vertical ? 4 : 60, component.X, "saved left position");
                Equal(vertical ? 1077 : 800, component.Y, "saved top position");
                Equal(vertical ? 1072 : 1800, component.Width, "saved width");
                Equal(vertical ? 313 : 230, component.Height, "saved height");
                Equal(50, component.Layer, "default layer");

                component.InteractionsTemplate = "invalid";
                component.InteractionsLayout = "invalid";
                component.InteractionsPageSeconds = -20;
                component.InteractionsSpeed = 999;
                component.InteractionsCardColor = "invalid";
                component.InteractionsCardNoise = 999;
                component.X = -20; component.Y = 9999;
                component.ShowTitle = true; component.ShowDetails = true;
                CreatorToolsOverlayComposerSettings.NormalizeComponent(profile, component);
                False(component.ShowTitle, "legacy scenes cannot enable interaction item names");
                False(component.ShowDetails, "legacy scenes cannot enable interaction details");
                Equal("floating", component.InteractionsTemplate, "invalid template falls back safely");
                Equal(vertical ? "pages" : "strip", component.InteractionsLayout, "invalid layout uses profile default");
                Equal(3, component.InteractionsPageSeconds, "normalization clamps minimum page duration");
                Equal(140, component.InteractionsSpeed, "normalization clamps maximum speed");
                Equal("#23262bee", component.InteractionsCardColor, "invalid card color falls back safely");
                Equal(100, component.InteractionsCardNoise, "normalization clamps maximum texture strength");
                Equal(0, component.X, "normalization clamps left edge");
                Equal(profile.CanvasHeight - component.Height, component.Y, "normalization clamps bottom edge");
            }
            var cloned = settings.Clone();
            cloned.FindProfile("vertical").FindComponent("interactions").InteractionsTikTok = true;
            cloned.FindProfile("vertical").FindComponent("interactions").InteractionsCardColor = "#aabbcc";
            cloned.FindProfile("vertical").FindComponent("interactions").InteractionsCardNoise = 0;
            False(settings.FindProfile("vertical").FindComponent("interactions").InteractionsTikTok,
                "candidate clone cannot mutate active profile");
            Equal("#23262bee", settings.FindProfile("vertical").FindComponent("interactions").InteractionsCardColor,
                "candidate clone cannot mutate the active card color");
            Equal(100, settings.FindProfile("vertical").FindComponent("interactions").InteractionsCardNoise,
                "candidate clone cannot mutate the active card texture");
        }

        private static void TestInteractionsOverlayPersistence()
        {
            using (var scope = new TestScope("interactions-overlay-persistence"))
            {
                var controller = scope.CreateOverlayComposer(null);
                var response = controller.ProcessConfigCommand(JsonSerializer.Serialize(new {
                    operation = "update", expectedRevision = 0, profileId = "horizontal", componentId = "interactions",
                    enabled = true, interactionsTemplate = "cards", interactionsLayout = "pages",
                    interactionsTikTok = false, interactionsTwitch = true, interactionsYouTube = false,
                    interactionsPageSeconds = 17, interactionsSpeed = 73, interactionsAdaptive = false, interactionsSize = 67,
                    interactionsCardColor = "#AABBCC80", interactionsCardNoise = 67,
                    x = 77, y = 850, width = 1700, height = 200, opacity = 87, locked = true
                }));
                Equal(200, response.StatusCode, "save complete interactions settings");
                controller = scope.CreateOverlayComposer(null);
                using (var state = JsonDocument.Parse(controller.GetConfigState()))
                {
                    var horizontal = FindComposerComponent(state.RootElement, "horizontal", "interactions");
                    AssertCustomizedInteractionsSettings(horizontal);
                    Equal(77, horizontal.GetProperty("x").GetInt32(), "position survives restart");
                    True(horizontal.GetProperty("locked").GetBoolean(), "lock survives restart");
                    Equal(87, horizontal.GetProperty("opacity").GetInt32(), "opacity survives restart");
                    var vertical = FindComposerComponent(state.RootElement, "vertical", "interactions");
                    True(vertical.GetProperty("enabled").GetBoolean(), "vertical activation stays independent");
                    False(vertical.GetProperty("interactionsTikTok").GetBoolean(), "vertical networks stay independent");
                    Equal(8, vertical.GetProperty("interactionsPageSeconds").GetInt32(), "vertical timing stays independent");
                    Equal("#662effa6", vertical.GetProperty("interactionsCardColor").GetString(), "vertical card color stays independent");
                    Equal(80, vertical.GetProperty("interactionsCardNoise").GetInt32(), "vertical texture stays independent");
                }
                response = controller.ProcessConfigCommand(JsonSerializer.Serialize(new {
                    operation = "copy", expectedRevision = 1, sourceProfileId = "horizontal",
                    profileId = "vertical", componentId = "interactions"
                }));
                Equal(200, response.StatusCode, "copy component settings across profiles");
                controller = scope.CreateOverlayComposer(null);
                using (var state = JsonDocument.Parse(controller.GetConfigState()))
                {
                    AssertCustomizedInteractionsSettings(FindComposerComponent(state.RootElement, "vertical", "interactions"));
                    AssertCustomizedInteractionsSettings(FindComposerComponent(state.RootElement, "horizontal", "interactions"));
                    False(FindComposerComponent(state.RootElement, "vertical", "roulette").GetProperty("enabled").GetBoolean(),
                        "copying interactions does not enable other overlays");
                }
                response = controller.ProcessConfigCommand(JsonSerializer.Serialize(new {
                    operation = "reset", expectedRevision = 2, profileId = "vertical", componentId = "interactions"
                }));
                Equal(200, response.StatusCode, "reset interactions only");
                using (var state = JsonDocument.Parse(response.Json))
                {
                    var vertical = FindComposerComponent(state.RootElement, "vertical", "interactions");
                    True(vertical.GetProperty("enabled").GetBoolean(), "reset restores saved activation");
                    Equal("floating", vertical.GetProperty("interactionsTemplate").GetString(), "reset restores template");
                    Equal("strip", vertical.GetProperty("interactionsLayout").GetString(), "reset restores profile layout");
                    False(vertical.GetProperty("interactionsTikTok").GetBoolean(), "reset restores TikTok");
                    True(vertical.GetProperty("interactionsTwitch").GetBoolean(), "reset restores Twitch for vertical");
                    True(vertical.GetProperty("interactionsAdaptive").GetBoolean(), "reset restores adaptive sizes");
                    Equal(8, vertical.GetProperty("interactionsPageSeconds").GetInt32(), "reset restores page timing");
                    Equal(80, vertical.GetProperty("interactionsSpeed").GetInt32(), "reset restores strip speed");
                    Equal(96, vertical.GetProperty("interactionsSize").GetInt32(), "reset restores saved size");
                    Equal("#662effa6", vertical.GetProperty("interactionsCardColor").GetString(), "reset restores card color");
                    Equal(80, vertical.GetProperty("interactionsCardNoise").GetInt32(), "reset restores card texture");
                    AssertCustomizedInteractionsSettings(FindComposerComponent(state.RootElement, "horizontal", "interactions"));
                }
                response = controller.ProcessConfigCommand(JsonSerializer.Serialize(new {
                    operation = "copy", expectedRevision = 3, sourceProfileId = "horizontal", profileId = "vertical"
                }));
                Equal(200, response.StatusCode, "whole profile copy includes interactions settings");
                using (var state = JsonDocument.Parse(response.Json))
                    AssertCustomizedInteractionsSettings(FindComposerComponent(state.RootElement, "vertical", "interactions"));
            }
        }

        private static void TestInteractionsOverlayValidation()
        {
            using (var scope = new TestScope("interactions-overlay-validation"))
            {
                var controller = scope.CreateOverlayComposer(null);
                Equal(200, controller.ProcessConfigCommand(JsonSerializer.Serialize(new {
                    operation = "update", expectedRevision = 0, profileId = "vertical", componentId = "interactions",
                    x = 91, interactionsTikTok = false, interactionsTwitch = false, interactionsYouTube = false,
                    interactionsPageSeconds = 3, interactionsSpeed = 140, interactionsSize = 25,
                    interactionsCardColor = "#ABCDEF", interactionsCardNoise = 0
                })).StatusCode, "bounds and an empty network filter are valid");
                var before = controller.GetConfigState();
                var path = Path.Combine(scope.ConfigDirectory, "mx.gilomx.cuphead.bossroulette.overlay-composer.json");
                var saved = File.ReadAllText(path);
                var invalidValues = new Dictionary<string, object>[] {
                    new Dictionary<string, object> { { "interactionsTemplate", "unknown" } },
                    new Dictionary<string, object> { { "interactionsLayout", "grid" } },
                    new Dictionary<string, object> { { "interactionsTikTok", "yes" } },
                    new Dictionary<string, object> { { "interactionsTwitch", 1 } },
                    new Dictionary<string, object> { { "interactionsYouTube", null } },
                    new Dictionary<string, object> { { "interactionsAdaptive", "enabled" } },
                    new Dictionary<string, object> { { "interactionsPageSeconds", 2 } },
                    new Dictionary<string, object> { { "interactionsPageSeconds", 61 } },
                    new Dictionary<string, object> { { "interactionsPageSeconds", 8.5 } },
                    new Dictionary<string, object> { { "interactionsSpeed", 9 } },
                    new Dictionary<string, object> { { "interactionsSpeed", 141 } },
                    new Dictionary<string, object> { { "interactionsSpeed", 73.5 } },
                    new Dictionary<string, object> { { "interactionsCardColor", "red" } },
                    new Dictionary<string, object> { { "interactionsCardColor", "#aabbccz1" } },
                    new Dictionary<string, object> { { "interactionsCardColor", "#abc" } },
                    new Dictionary<string, object> { { "interactionsCardColor", null } },
                    new Dictionary<string, object> { { "interactionsCardNoise", -1 } },
                    new Dictionary<string, object> { { "interactionsCardNoise", 101 } },
                    new Dictionary<string, object> { { "interactionsCardNoise", 18.5 } },
                    new Dictionary<string, object> { { "interactionsCardNoise", "many" } },
                    new Dictionary<string, object> { { "interactionsSize", -1 } },
                    new Dictionary<string, object> { { "interactionsSize", 24 } },
                    new Dictionary<string, object> { { "interactionsSize", 101 } },
                    new Dictionary<string, object> { { "interactionsSize", 75.5 } },
                    new Dictionary<string, object> { { "interactionsSize", "big" } }
                };
                foreach (var invalid in invalidValues)
                {
                    var command = new Dictionary<string, object>(invalid) {
                        { "operation", "update" }, { "expectedRevision", 1 }, { "profileId", "vertical" },
                        { "componentId", "interactions" }, { "x", 777 }, { "enabled", true }
                    };
                    Equal(400, controller.ProcessConfigCommand(JsonSerializer.Serialize(command)).StatusCode,
                        "invalid interactions field rejected");
                    Equal(before, controller.GetConfigState(), "invalid request preserves active scene and revision");
                    Equal(saved, File.ReadAllText(path), "invalid request never writes persistent scene");

                    var invalidFile = JsonNode.Parse(saved);
                    var profileNode = invalidFile["profiles"][0];
                    var componentNode = profileNode["components"].AsArray().First(item => item["id"].GetValue<string>() == "interactions");
                    foreach (var pair in invalid)
                        componentNode[pair.Key] = JsonSerializer.SerializeToNode(pair.Value);
                    CreatorToolsOverlayComposerProfile parsed;
                    False(CreatorToolsOverlayComposerSettings.TryParseProfileJson(profileNode.ToJsonString(), "vertical", out parsed),
                        "preview rejects invalid interactions field");
                    File.WriteAllText(path + ".bak", saved);
                    File.WriteAllText(path, invalidFile.ToJsonString());
                    Equal(before, scope.CreateOverlayComposer(null).GetConfigState(), "invalid stored field recovers the valid backup");
                    Equal(saved, File.ReadAllText(path), "backup recovery restores the primary unchanged");
                    Equal(saved, File.ReadAllText(path + ".bak"), "backup recovery preserves the recovery copy");
                }
                foreach (var oldSpeed in new[] { 10, 39 })
                {
                    Equal(400, controller.ProcessConfigCommand(JsonSerializer.Serialize(new {
                        operation = "update", expectedRevision = 1, profileId = "vertical", componentId = "interactions",
                        x = 800, interactionsSpeed = oldSpeed
                    })).StatusCode, "API rejects speeds below its new minimum");
                    Equal(before, controller.GetConfigState(), "rejected legacy API speed preserves the active scene");
                    Equal(saved, File.ReadAllText(path), "rejected legacy API speed preserves saved settings");
                }
                foreach (var pair in new Dictionary<string, object> {
                    { "interactionsSpeed", 40 }, { "interactionsCardColor", "#123456" }, { "interactionsCardNoise", 0 }
                })
                {
                    var command = new Dictionary<string, object> {
                        { "operation", "update" }, { "expectedRevision", 1 }, { "profileId", "vertical" },
                        { "componentId", "tap_farming" }, { "x", 800 }, { pair.Key, pair.Value }
                    };
                    Equal(400, controller.ProcessConfigCommand(JsonSerializer.Serialize(command)).StatusCode,
                        "interactions fields cannot mutate another overlay");
                }
                Equal(200, controller.ProcessConfigCommand(JsonSerializer.Serialize(new {
                    operation = "update", expectedRevision = 1, profileId = "vertical", componentId = "interactions",
                    interactionsPageSeconds = 60, interactionsSpeed = 40, interactionsSize = 100,
                    interactionsCardNoise = 100
                })).StatusCode, "other range boundaries are accepted");
                using (var state = JsonDocument.Parse(controller.GetConfigState()))
                {
                    var current = FindComposerComponent(state.RootElement, "vertical", "interactions");
                    Equal("#abcdef", current.GetProperty("interactionsCardColor").GetString(), "six digit colors retain their opacity semantics");
                    Equal(100, current.GetProperty("interactionsCardNoise").GetInt32(), "maximum texture strength is accepted");
                    False(FindComposerComponent(state.RootElement, "vertical", "roulette").TryGetProperty("interactionsCardColor", out _),
                        "unrelated overlays expose no interactions card fields");
                }
                Equal(200, controller.ProcessConfigCommand(JsonSerializer.Serialize(new {
                    operation = "update", expectedRevision = 2, profileId = "vertical", componentId = "interactions",
                    interactionsSize = 0
                })).StatusCode, "automatic size can be restored");
                using (var state = JsonDocument.Parse(controller.GetConfigState()))
                    Equal(0, FindComposerComponent(state.RootElement, "vertical", "interactions").GetProperty("interactionsSize").GetInt32(),
                        "automatic size persists after a manual setting");
            }
        }

        private static void TestInteractionsOverlayMigration()
        {
            using (var scope = new TestScope("interactions-overlay-migration"))
            {
                var controller = scope.CreateOverlayComposer(null);
                True(controller.ImportRouletteSettings(false, 150, "right", 79, true, true, "keep"), "mark roulette settings imported");
                Equal(200, controller.ProcessConfigCommand(JsonSerializer.Serialize(new {
                    operation = "update", expectedRevision = 1, profileId = "vertical", componentId = "pesky_battle",
                    x = 97, opacity = 63, locked = true, textColor = "#12345680"
                })).StatusCode, "save a customized existing scene");
                var path = Path.Combine(scope.ConfigDirectory, "mx.gilomx.cuphead.bossroulette.overlay-composer.json");
                var legacy = JsonNode.Parse(File.ReadAllText(path));
                foreach (var profile in legacy["profiles"].AsArray())
                {
                    var components = profile["components"].AsArray();
                    components.Remove(components.First(item => item["id"].GetValue<string>() == "interactions"));
                    CreatorToolsOverlayComposerProfile preview;
                    True(CreatorToolsOverlayComposerSettings.TryParseProfileJson(profile.ToJsonString(), profile["id"].GetValue<string>(), out preview),
                        "legacy layout without interactions remains a valid preview");
                    False(preview.FindComponent("interactions").Enabled, "legacy preview never enables a newly added interaction layer");
                }
                var legacyJson = legacy.ToJsonString();
                File.WriteAllText(path, legacyJson);
                controller = scope.CreateOverlayComposer(null);
                Equal(legacyJson, File.ReadAllText(path + ".bak"), "migration retains original scene as a recovery backup");
                using (var state = JsonDocument.Parse(controller.GetConfigState()))
                {
                    Equal(2, state.RootElement.GetProperty("revision").GetInt32(), "migration does not invalidate saved revision");
                    foreach (var oldProfile in legacy["profiles"].AsArray())
                    {
                        var profileId = oldProfile["id"].GetValue<string>();
                        foreach (var oldComponent in oldProfile["components"].AsArray())
                        {
                            var current = FindComposerComponent(state.RootElement, profileId, oldComponent["id"].GetValue<string>());
                            Equal(oldComponent.ToJsonString(), JsonNode.Parse(current.GetRawText()).ToJsonString(),
                                "migration retains every saved component field");
                        }
                        False(FindComposerComponent(state.RootElement, profileId, "interactions").GetProperty("enabled").GetBoolean(),
                            "migration keeps interactions invisible in OBS");
                    }
                }
                foreach (var profile in JsonNode.Parse(File.ReadAllText(path))["profiles"].AsArray())
                {
                    var parsedJson = profile.ToJsonString();
                    var component = profile["components"].AsArray().First(item => item["id"].GetValue<string>() == "interactions");
                    foreach (var key in component.AsObject().Select(pair => pair.Key).Where(key => key.StartsWith("interactions", StringComparison.Ordinal)).ToArray())
                        component.AsObject().Remove(key);
                    CreatorToolsOverlayComposerProfile parsed;
                    True(CreatorToolsOverlayComposerSettings.TryParseProfileJson(profile.ToJsonString(), profile["id"].GetValue<string>(), out parsed),
                        "older partial component safely receives optional defaults");
                    False(parsed.FindComponent("interactions").Enabled, "partial legacy component remains disabled");
                    var vertical = parsed.Id == "vertical";
                    Equal(vertical ? 96 : 0, parsed.FindComponent("interactions").InteractionsSize, "missing size receives the profile default");
                    Equal(vertical ? "#662effa6" : "#23262bee", parsed.FindComponent("interactions").InteractionsCardColor, "old scenes receive the default card color");
                    Equal(vertical ? 80 : 18, parsed.FindComponent("interactions").InteractionsCardNoise, "old scenes receive the default card texture");
                    Equal("strip",
                        parsed.FindComponent("interactions").InteractionsLayout, "missing fields retain per-profile layout defaults");
                    True(CreatorToolsOverlayComposerSettings.TryParseProfileJson(parsedJson, parsed.Id, out parsed),
                        "current component round trips through preview parser");
                }
                var persisted = File.ReadAllText(path);
                using (var ignored = JsonDocument.Parse(scope.CreateOverlayComposer(null).GetConfigState())) { }
                Equal(persisted, File.ReadAllText(path), "subsequent load does not rewrite migrated file");
                Equal(legacyJson, File.ReadAllText(path + ".bak"), "subsequent load retains original recovery backup");
            }
        }

        private static void TestInteractionsOverlaySpeedMigration()
        {
            foreach (var speed in new[] { 10, 39, 40, 120, 140 })
            using (var scope = new TestScope("interactions-overlay-speed-migration-" + speed))
            {
                var controller = scope.CreateOverlayComposer(null);
                True(controller.ImportRouletteSettings(false, 150, "right", 79, true, true, "keep"), "mark roulette imported before saving a legacy scene");
                Equal(200, controller.ProcessConfigCommand(JsonSerializer.Serialize(new {
                    operation = "update", expectedRevision = 1, profileId = "vertical", componentId = "interactions",
                    enabled = true, locked = true, x = 91, y = 1170, opacity = 63, interactionsTemplate = "cards",
                    interactionsSpeed = 73, interactionsSize = 67, interactionsAdaptive = false,
                    interactionsTikTok = false, interactionsTwitch = true, interactionsYouTube = false
                })).StatusCode, "save customized interactions before migration");
                var path = Path.Combine(scope.ConfigDirectory, "mx.gilomx.cuphead.bossroulette.overlay-composer.json");
                var legacy = JsonNode.Parse(File.ReadAllText(path));
                foreach (var profile in legacy["profiles"].AsArray())
                {
                    var component = profile["components"].AsArray().First(item => item["id"].GetValue<string>() == "interactions");
                    component["interactionsSpeed"] = speed;
                    component.AsObject().Remove("interactionsCardColor");
                    component.AsObject().Remove("interactionsCardNoise");
                    CreatorToolsOverlayComposerProfile preview;
                    True(CreatorToolsOverlayComposerSettings.TryParseProfileJson(profile.ToJsonString(), profile["id"].GetValue<string>(), out preview),
                        "preview accepts legacy speed and absent texture fields");
                    Equal(Math.Max(40, speed), preview.FindComponent("interactions").InteractionsSpeed,
                        "preview migrates only speeds below the new minimum");
                    Equal(preview.Id == "vertical" ? "#662effa6" : "#23262bee", preview.FindComponent("interactions").InteractionsCardColor, "preview receives the default card color");
                    Equal(preview.Id == "vertical" ? 80 : 18, preview.FindComponent("interactions").InteractionsCardNoise, "preview receives the default texture strength");
                }
                var legacyJson = legacy.ToJsonString();
                File.WriteAllText(path, legacyJson);
                controller = scope.CreateOverlayComposer(null);
                Equal(legacyJson, File.ReadAllText(path + ".bak"), "migration preserves the complete old scene in its backup");
                var expected = JsonNode.Parse(legacyJson);
                foreach (var profile in expected["profiles"].AsArray())
                {
                    var component = profile["components"].AsArray().First(item => item["id"].GetValue<string>() == "interactions");
                    component["interactionsSpeed"] = Math.Max(40, speed);
                    component["interactionsCardColor"] = profile["id"].GetValue<string>() == "vertical" ? "#662effa6" : "#23262bee";
                    component["interactionsCardNoise"] = profile["id"].GetValue<string>() == "vertical" ? 80 : 18;
                }
                using (var state = JsonDocument.Parse(controller.GetConfigState()))
                {
                    Equal(2, state.RootElement.GetProperty("revision").GetInt32(), "speed migration retains the existing revision");
                    True(JsonNode.DeepEquals(expected["profiles"], JsonNode.Parse(state.RootElement.GetProperty("profiles").GetRawText())),
                        "speed migration retains all unrelated fields, geometry, visibility and components");
                }
                var persisted = File.ReadAllText(path);
                using (var current = JsonDocument.Parse(scope.CreateOverlayComposer(null).GetConfigState()))
                    Equal(Math.Max(40, speed), FindComposerComponent(current.RootElement, "vertical", "interactions").GetProperty("interactionsSpeed").GetInt32(),
                        "migrated speed survives another load");
                Equal(persisted, File.ReadAllText(path), "migration persists once without rewriting on later loads");
                Equal(legacyJson, File.ReadAllText(path + ".bak"), "later loads retain the pre-migration recovery backup");
            }
        }
    }
