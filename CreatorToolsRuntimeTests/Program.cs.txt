        private static void TestRuleCooldownScheduling()
        {
            using (var scope = new TestScope("rule-cooldowns"))
            {
                var now = new DateTime(
                    2026, 9, 27, 12, 0, 0, DateTimeKind.Utc);
                var rules = scope.CreateRules(
                    1, 10, 4, delegate { return now; });
                var gameplay = new CreatorToolsInteractionController(200);

                rules.Evaluate(NewGiftEvent(1, 1), true);
                Equal(1, rules.Update(null, gameplay, true),
                    "first redeem drained");
                Equal(0f, gameplay.LastDelaySeconds,
                    "first redeem has no delay");

                now = now.AddSeconds(1);
                rules.Evaluate(NewGiftEvent(1, 1), true);
                Equal(1L, rules.BacklogCount,
                    "same-user redeem retained during cooldown");
                Equal(1, rules.Update(null, gameplay, true),
                    "same-user redeem drained into scheduled queue");
                True(Math.Abs(gameplay.LastDelaySeconds - 9f) < 0.01f,
                    "per-user cooldown reserves the next slot");

                rules.Evaluate(NewGiftEvent(2, 1), true);
                Equal(1, rules.Update(null, gameplay, true),
                    "other-user redeem also retained");
                True(Math.Abs(gameplay.LastDelaySeconds - 13f) < 0.01f,
                    "shared gift cooldown follows the reserved slot");
                Equal(0L, rules.BacklogCount,
                    "all cooldown redeems materialized without loss");
            }
        }

        private static void TestRuleCooldownBatchScheduling()
        {
            using (var scope = new TestScope("rule-cooldown-batch"))
            {
                var now = new DateTime(
                    2026, 9, 27, 13, 0, 0, DateTimeKind.Utc);
                var rules = scope.CreateRules(
                    2, 10, 4, delegate { return now; });
                var gameplay = new CreatorToolsInteractionController(200);

                var result = rules.Evaluate(NewGiftEvent(7, 30), true);
                Equal(60L, result.DeferredInteractions,
                    "thirty gifts preserve two interactions per activation");
                Equal(60L, rules.BacklogCount,
                    "all interactions remain in the backlog before drain");

                var drained = 0;
                var updates = 0;
                while (rules.BacklogCount > 0 && updates < 10)
                {
                    drained += rules.Update(null, gameplay, true);
                    updates++;
                }
                Equal(60, drained,
                    "all scheduled batches materialize into the gameplay queue");
                Equal(30, gameplay.EnqueueCallCount,
                    "one enqueue batch is retained per gift activation");
                Equal(30, gameplay.EnqueueDelays.Count,
                    "every activation publishes its own due time");
                True(Math.Abs(gameplay.EnqueueDelays[0] - 0f) < 0.01f,
                    "first activation is immediate");
                True(Math.Abs(gameplay.EnqueueDelays[1] - 10f) < 0.01f,
                    "second activation waits for the effective cooldown");
                True(Math.Abs(gameplay.EnqueueDelays[29] - 290f) < 0.01f,
                    "thirtieth activation reserves the final cooldown slot");
                Equal(0L, rules.BacklogCount,
                    "the serialized batch loses no interactions");
            }
        }

        private static void TestTapFarmingEffectiveHealth()
        {
            var collecting =
                CreatorToolsTapFarmingEffectiveHealth.Calculate(
                    "collecting", 0d, 0d, 500d, 0d);
            False(collecting.Available,
                "collecting has no percentage denominator");
            Equal(500d, collecting.Current,
                "collecting exposes accumulated health points");

            var initial =
                CreatorToolsTapFarmingEffectiveHealth.Calculate(
                    "active", 1000d, 1000d, 500d, 0d);
            True(initial.Available, "active health is available");
            Equal(1500d, initial.Current,
                "initial effective current health");
            Equal(1500d, initial.Total,
                "initial effective total health");
            Equal(1d, initial.Ratio,
                "initial effective ratio");

            var absorbed =
                CreatorToolsTapFarmingEffectiveHealth.Calculate(
                    "active", 1000d, 1000d, 400d, 100d);
            Equal(1400d, absorbed.Current,
                "absorbed damage reduces effective current health");
            Equal(1500d, absorbed.Total,
                "absorbed damage preserves effective total health");
            True(Math.Abs(absorbed.Ratio - (1400d / 1500d)) < 0.000001d,
                "absorbed damage changes the ratio before native health");

            var nativeDamage =
                CreatorToolsTapFarmingEffectiveHealth.Calculate(
                    "active", 900d, 1000d, 0d, 500d);
            Equal(900d, nativeDamage.Current,
                "native damage continues reducing effective health");
            Equal(1500d, nativeDamage.Total,
                "effective total includes consumed reserve");

            var completed =
                CreatorToolsTapFarmingEffectiveHealth.Calculate(
                    "completed", 0d, 1000d, 125d, 375d);
            True(completed.Available,
                "completed state keeps a percentage result");
            Equal(0d, completed.Current,
                "completed state ignores leftover reserve");
            Equal(0d, completed.Ratio,
                "completed state is empty");
        }

        private static void TestDashboardChatSimulationVoting()
        {
            using (var scope = new TestScope("chat-vote-simulator"))
            using (var server = scope.StartServer())
            {
                var clock = DateTime.UtcNow.AddMinutes(-1);
                var controller = new CreatorToolsChatChoosesController(
                    new CreatorToolsLiveEventsCoordinator(), true, new Random(1), delegate { return clock; });
                var catalog = new ChatChoosesCatalog();
                catalog.Pools["boss"] = new List<ChatChoice> {
                    new ChatChoice { Id = 1 }, new ChatChoice { Id = 2 } };
                controller.SetCatalog(catalog, true);
                server.SetChatChoosesController(controller, controller.Command);
                Equal(HttpStatusCode.Accepted, GetStatusCode(server,
                    "/api/config/chat-chooses/set?operation=start&mode=with"), "open real voting controller");
                var dashboard = new CreatorToolsDashboardController(null);
                var warnings = 0;
                using (var worker = new CreatorToolsStreamWorker(null, dashboard, null, server,
                    delegate(CreatorToolsStreamEvent entry) {
                        return new CreatorToolsStreamEvaluation { MessageCode = controller.Observe(entry) };
                    }, null, delegate { Interlocked.Increment(ref warnings); }))
                {
                    server.SetDashboardSimulationHandler(worker.ScheduleSimulation);
                    worker.Start();
                    var requests = new[] { "userId=sim-cuphead&chatText=1", "userId=sim-cuphead&chatText=1",
                        "userId=sim-cuphead&chatText=2", "userId=sim-mugman&chatText=2" };
                    var totals = new[] { 1, 1, 1, 2 };
                    for (var i = 0; i < requests.Length; i++)
                    {
                        Equal(HttpStatusCode.Accepted, GetStatusCode(server,
                            "/api/dashboard/simulate?platform=tiktok&type=chat&count=99&amount=100&" + requests[i]), "HTTP simulation accepted");
                        var expectedReceived = i + 1;
                        WaitUntil(delegate {
                            using (var feed = GetJson(server, "/api/dashboard"))
                                return feed.RootElement.GetProperty("counters").GetProperty("received").GetInt64() == expectedReceived;
                        }, 3000, "background worker did not publish the ballot");
                        using (var state = GetJson(server, "/api/config/chat-chooses"))
                            Equal(totals[i], state.RootElement.GetProperty("totalVotes").GetInt32(), "one vote per identity regardless of quantity");
                        using (var feed = GetJson(server, "/api/dashboard"))
                        {
                            var entry = feed.RootElement.GetProperty("events")[0];
                            Equal(i == 1 ? "chat_vote_unchanged" : "chat_vote_counted", entry.GetProperty("messageCode").GetString(), "ballot feedback published");
                            Equal(1, entry.GetProperty("count").GetInt32(), "chat quantity normalized");
                            Equal(0m, entry.GetProperty("totalValue").GetDecimal(), "chat has no currency value");
                        }
                    }
                    using (var state = GetJson(server, "/api/config/chat-chooses"))
                    {

                    {

                    });

                worker.Start();

                    {
                        return source.TakenCount == 1 &&
                            rules.BacklogCount == 2;
                    },

                    {
                        using (var json = GetJson(
                            server, "/api/config/interactions"))
                            return json.RootElement
                                .GetProperty("backlogCount").GetInt64() == 2L;
                    },
                    3000,
                    "worker did not publish the backlog projection");

                {
                    Equal(1L,
                        json.RootElement.GetProperty("counters")
                            .GetProperty("received").GetInt64(),
                        "worker-published dashboard received count");

                    "background-state callback was not invoked");

                {
                    Event = NewGiftEvent(100, 1)
                });
                worker.Signal();
                Thread.Sleep(150);

                    "source updates after Dispose returned");

                    Equal("Bomba", json.RootElement.GetProperty("attackName").GetString(),
                        "attack preview display name");
                    Equal("/assets/bomb.png", json.RootElement.GetProperty("attackImagePath").GetString(),
                        "attack preview image path");
                    Equal(1790985600000L, json.RootElement.GetProperty("attackStartedAt").GetInt64(),
                        "attack timestamp survives native serialization");
                    Equal(3, json.RootElement.GetProperty("battleSignals").GetProperty("attacks").GetArrayLength(),
                        "OBS transport preserves a whole burst between polls");
                    Equal("countdown", json.RootElement.GetProperty("battleSignals").GetProperty("challenges")[0].GetProperty("phase").GetString(),
                        "OBS transport includes persistent challenge state");
                    Equal(7, json.RootElement.GetProperty("eventEpoch").GetInt32(), "preview event reset generation");
                }
                foreach (var invalidSignals in new[] {
                    "{\"attacks\":[],\"challenges\":{}}",
                    battleSignalsJson.Replace("\"slot\":2", "\"slot\":6"),
                    battleSignalsJson.Replace("\"slot\":2", "\"slot\":2.5"),
                    battleSignalsJson.Replace("\"id\":\"a2\"", "\"id\":\"a1\""),
                    battleSignalsJson.Replace("\"secondsRemaining\":3", "\"secondsRemaining\":121"),
                    battleSignalsJson.Replace("\"phase\":\"countdown\"", "\"phase\":\"off\"")
                })
                {
                    var rejected = controller.ProcessPreviewCommand(JsonSerializer.Serialize(new {
                        operation = "update", profileId = "horizontal", sessionId = "battle-editor-a",
                        attackName = "Must not replace", battleSignalsJson = invalidSignals
                    }));
                    Equal(400, rejected.StatusCode, "malformed preview signals are rejected");
                    using (var json = JsonDocument.Parse(controller.GetPreviewState("horizontal").Json))
                    {
                        Equal(firstBattleRevision, json.RootElement.GetProperty("revision").GetInt32(), "invalid signals leave preview revision unchanged");
                        Equal("Bomba", json.RootElement.GetProperty("attackName").GetString(), "invalid signals leave the whole preview unchanged");
                    }
                }
                var replacementPreview = controller.ProcessPreviewCommand(
                    "{\"operation\":\"start\"," +
                    "\"profileId\":\"horizontal\"," +
                    "\"sessionId\":\"battle-editor-b\"," +
                    "\"componentId\":\"pesky_battle\"}");
                int replacementRevision;
                int replacementRunId;
                using (var json = JsonDocument.Parse(
                    replacementPreview.Json))
                {
                    replacementRevision = json.RootElement
                        .GetProperty("revision").GetInt32();
                    replacementRunId = json.RootElement
                        .GetProperty("runId").GetInt32();
                    Equal("", json.RootElement.GetProperty("attackId").GetString(),
                        "replacement preview cannot inherit an old attack");
                }
                True(replacementRunId > firstBattleRunId,
                    "replacement preview receives a new run id");
                var staleStop = controller.ProcessPreviewCommand(
                    "{\"operation\":\"stop\"," +
                    "\"profileId\":\"horizontal\"," +
                    "\"sessionId\":\"battle-editor-a\"," +
                    "\"expectedRevision\":" + firstBattleRevision + "}");
                Equal(409, staleStop.StatusCode,
                    "stale preview cannot stop replacement");
                var stopped = controller.ProcessPreviewCommand(
                    "{\"operation\":\"stop\"," +
                    "\"profileId\":\"horizontal\"," +
                    "\"sessionId\":\"battle-editor-b\"," +
                    "\"expectedRevision\":" + replacementRevision + "}");
                using (var json = JsonDocument.Parse(stopped.Json))
                {
                    False(json.RootElement
                        .GetProperty("active").GetBoolean(),
                        "stopped preview active flag");
                    Equal("stopped", json.RootElement
                        .GetProperty("feedback").GetString(),
                        "stopped preview feedback");
                }
            }
        }

        private static JsonElement FindComposerComponent(
            JsonElement root, string profileId, string componentId)
        {
            foreach (var profile in root.GetProperty("profiles")
                         .EnumerateArray())
            {
                if (profile.GetProperty("id").GetString() != profileId)
                    continue;
                foreach (var component in profile
                             .GetProperty("components").EnumerateArray())
                    if (component.GetProperty("id").GetString() ==
                        componentId)
                        return component;
            }
            throw new InvalidOperationException(
                "composer component was not found");
        }

            JsonElement profile, string componentId)

            foreach (var component in profile.GetProperty("components")
                         .EnumerateArray())
                if (component.GetProperty("id").GetString() == componentId)
                    return component;
            throw new InvalidOperationException(
                "profile component was not found");
        }

        private static void TestTikFinityLikePressureCoalescing()
        {
            using (var host = NewCompanionHost())
            {
                FillCompanionQueueWithFollows(host);
                var first = NewCompanionLike(
                    "pressure-a", "viewer-pressure", 700000);
                var second = NewCompanionLike(
                    "pressure-b", "viewer-pressure", 700000);
                EnqueueCompanion(host, first);
                EnqueueCompanion(host, second);
                // The transport coalescer runs before dashboard dedupe. A
                // resent source identity must not be added a second time.
                EnqueueCompanion(host, first);

                long likeCount = 0L;
                var likeMessages = 0;
                var aggregateIds = new HashSet<string>(
                    StringComparer.Ordinal);
                CreatorToolsStreamMessage message;
                while (host.TryTakeMessage(out message))
                {
                    if (message.Event == null ||
                        message.Event.Type != "like")
                        continue;
                    likeMessages++;
                    likeCount += message.Event.Count;
                    True(message.Event.Count <= 1000000,
                        "coalesced like exceeded dashboard count limit");
                    Equal("viewer-pressure", message.Event.UserId,
                        "coalesced viewer identity");
                    True(aggregateIds.Add(message.Event.IdempotencyKey),
                        "coalesced chunks reused an idempotency key");
                }

                Equal(1400000L, likeCount,
                    "coalesced per-viewer like total");
                Equal(2, likeMessages,
                    "coalesced count chunk quantity");
            }
        }

        private static void TestTikFinityCriticalEventPreservesLikes()
        {
            using (var host = NewCompanionHost())
            {
                const int capacity = 1024;
                for (var i = 0; i < capacity; i++)
                    EnqueueCompanion(host, NewCompanionLike(
                        "evicted-" + i.ToString(
                            CultureInfo.InvariantCulture),

                            CultureInfo.InvariantCulture),

                CreatorToolsStreamMessage message;
                while (host.TryTakeMessage(out message))

                }

                Equal(2048L, likeCount,
                    "likes retained after critical eviction");
                Equal(1, criticalGifts,
                    "critical gift retained under pressure");
            }
        }

        private static void TestTikFinityLikeOverflowFallback()
        {
            using (var host = NewCompanionHost())
            {
                FillCompanionQueueWithFollows(host);
                const int distinctViewers = 4097;
                for (var i = 0; i < distinctViewers; i++)
                    EnqueueCompanion(host, NewCompanionLike(
                        "fallback-" + i.ToString(

                    "anonymous overflow fallback count");
            }
        }

        private static JsonDocument PostJson(
            CreatorToolsServer server,
            string path,
            string body,
            out HttpStatusCode status)
        {
            using (var client = new HttpClient())
            using (var content = new StringContent(
                body, Encoding.UTF8, "application/json"))
            using (var response = client.PostAsync(
                "http://127.0.0.1:" +
                server.Port.ToString(CultureInfo.InvariantCulture) + path,
                content).GetAwaiter().GetResult())
            {
                status = response.StatusCode;
                var text = response.Content.ReadAsStringAsync()
                    .GetAwaiter().GetResult();
                return JsonDocument.Parse(text);
            }
        }

        private static bool IsUnityAssemblyLoaded()
        {
            var assemblies = AppDomain.CurrentDomain.GetAssemblies();
            for (var i = 0; i < assemblies.Length; i++)
            {
                var name = assemblies[i].GetName().Name ?? string.Empty;
                if (name.StartsWith("UnityEngine", StringComparison.Ordinal))
                    return true;
            }
            return false;
        }

        private static void WaitUntil(
            Func<bool> condition, int timeoutMilliseconds, string failure)
        {
            var expires = Environment.TickCount64 + timeoutMilliseconds;
            Exception lastError = null;
            while (Environment.TickCount64 < expires)
            {
                try
                {
                    if (condition())
                        return;
                }
                catch (Exception exception)
                {
                    lastError = exception;
                }
                Thread.Sleep(20);
            }
            throw new InvalidOperationException(
                failure + (lastError == null
                    ? string.Empty
                    : ": " + lastError.Message));
        }

                throw new InvalidOperationException(
                    label + ": expected " + expected + ", got " + actual);
        }

        private sealed class FakeStreamSource : ICreatorToolsStreamSource
        {
            private readonly object queueLock = new object();
            private readonly Queue<CreatorToolsStreamMessage> messages =
                new Queue<CreatorToolsStreamMessage>();
            private int updateCount;
            private int takenCount;

            internal int UpdateCount
            {
                get { return Interlocked.CompareExchange(
                    ref updateCount, 0, 0); }
            }

            internal int TakenCount
            {
                get { return Interlocked.CompareExchange(
                    ref takenCount, 0, 0); }
            }

            internal void Enqueue(CreatorToolsStreamMessage message)
            {
                lock (queueLock)
                    messages.Enqueue(message);
            }

            public void Update()
            {
                Interlocked.Increment(ref updateCount);
            }

            public bool TryTakeMessage(
                out CreatorToolsStreamMessage message)
            {
                lock (queueLock)
                {
                    if (messages.Count == 0)
                    {
                        message = null;
                        return false;
                    }
                    message = messages.Dequeue();
                }
                Interlocked.Increment(ref takenCount);
                return true;
            }
        }
